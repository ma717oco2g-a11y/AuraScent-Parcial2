-- 1. Tabla de Familias Olfativas
CREATE TABLE familias_olfativas (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL
);

-- 2. Tabla Principal de Perfumes
CREATE TABLE perfumes (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    marca VARCHAR(100) NOT NULL,
    concentracion VARCHAR(50),
    nota_salida VARCHAR(100),
    nota_corazon VARCHAR(100),
    nota_fondo VARCHAR(100),
    ocasion VARCHAR(50), 
    vibra VARCHAR(50),   
    familia_id INT,
    FOREIGN KEY (familia_id) REFERENCES familias_olfativas(id)
);
